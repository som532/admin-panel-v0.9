local TITLE = "Admin panel 0.9"

local function notify(t, x)
    pcall(function()
        SG:SetCore("SendNotification", {
            Title = t or TITLE,
            Text = x or "",
            Duration = 3
        })
    end)
end
Y.Notify = notify

function Y:FindPlayer(q)
    if not q or q == "" then return nil end
    q = q:lower()

    if q == "me" then return LP end
    if q == "random" then
        local list = {}
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= LP and p.Character then table.insert(list, p) end
        end
        if #list == 0 then return nil end
        return list[math.random(1, #list)]
    end

    for _, p in ipairs(Players:GetPlayers()) do
        if p.Name:lower() == q then return p end
    end
    for _, p in ipairs(Players:GetPlayers()) do
        if p.Name:lower():find(q, 1, true) then return p end
    end
end

function Y:Register(name, data)
    if type(name) ~= "string" or type(data) ~= "table" or type(data.Func) ~= "function" then return end
    data.Aliases = data.Aliases or {}
    self.Commands[name:lower()] = data
    for _, a in ipairs(data.Aliases) do self.Aliases[a:lower()] = name:lower() end
end

function Y:Get(name)
    name = (name or ""):lower()
    return self.Commands[name] or self.Commands[self.Aliases[name] or ""]
end

function Y:Run(input)
    if not input or input == "" then return end
    local args = {}
    for w in input:gmatch("%S+") do table.insert(args, w) end
    local name = table.remove(args, 1)
    local low = name:lower()

    if low:sub(1, 2) == "un" and #low > 2 then
        local base = self:Get(low:sub(3))
        if base and base.Unfunc then
            local ok, err = pcall(base.Unfunc, args, base)
            if not ok then notify("Error", tostring(err)) end
            return
        end
    end

    local cmd = self:Get(name)
    if not cmd then return notify(TITLE, "Unknown: " .. name) end
    local ok, err = pcall(cmd.Func, args, cmd)
    if not ok then notify("Error", tostring(err)) end
end

local function refresh(list, status, query)
    for _, c in ipairs(list:GetChildren()) do
        if c:IsA("TextButton") then c:Destroy() end
    end
    query = (query or ""):lower()
    local count = 0
    local sorted = {}
    for n, d in pairs(Y.Commands) do table.insert(sorted, {n = n, d = d}) end
    table.sort(sorted, function(a, b) return a.n < b.n end)

    for _, e in ipairs(sorted) do
        if query == "" or e.n:find(query, 1, true) then
            count = count + 1
            local btn = Instance.new("TextButton")
            btn.Size = UDim2.new(1, -12, 0, 34)
            btn.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
            btn.BorderSizePixel = 0
            btn.Text = ""
            btn.Parent = list
            Instance.new("UICorner", btn).CornerRadius = UDim.new(0, 5)

            local stroke = Instance.new("UIStroke")
            stroke.Color = Color3.fromRGB(200, 200, 200)
            stroke.Thickness = 1
            stroke.Parent = btn

            local lbl = Instance.new("TextLabel")
            lbl.Size = UDim2.new(1, -16, 1, 0)
            lbl.Position = UDim2.new(0, 8, 0, 0)
            lbl.BackgroundTransparency = 1
            lbl.Text = e.n
            lbl.TextColor3 = Color3.fromRGB(0, 0, 0)
            lbl.TextXAlignment = Enum.TextXAlignment.Left
            lbl.Font = Enum.Font.GothamBold
            lbl.TextSize = 14
            lbl.Parent = btn

            btn.MouseButton1Click:Connect(function()
                local b = Y.Gui and Y.Gui.Main and Y.Gui.Main:FindFirstChild("SearchBox")
                if b then
                    b.Text = e.n .. " "
                    b:CaptureFocus()
                end
            end)
        end
    end
    status.Text = "Found: " .. count
end

local function init()
    local gui, box, list, status = buildGui()
    Y.Gui = gui
    box:GetPropertyChangedSignal("Text"):Connect(function() refresh(list, status, box.Text) end)
    box.FocusLost:Connect(function(enter)
        if enter then
            local t = box.Text
            box.Text = ""
            refresh(list, status, "")
            if t ~= "" then Y:Run(t) end
        end
    end)
    refresh(list, status, "")
end

UIS.InputBegan:Connect(function(input, gp)
    if gp then return end
    if input.KeyCode == Enum.KeyCode.RightShift and Y.Gui then
        local m = Y.Gui:FindFirstChild("Main")
        if m then
            m.Visible = not m.Visible
            if m.Visible then
                local b = m:FindFirstChild("SearchBox")
                if b then b:CaptureFocus() end
            end
        end
    end
end)

Y:Register("speed", {
    Func = function(a)
        local n = tonumber(a[1])
        if not n then return notify("Speed", "Enter a number") end
        local h = LP.Character and LP.Character:FindFirstChildOfClass("Humanoid")
        if h then
            h.WalkSpeed = n
            notify("Speed", "Speed: " .. n)
        end
    end
})

Y:Register("loopsped", {
    Aliases = {"ls"},
    Func = function(a)
        local n = tonumber(a[1])
        if not n then return notify("LoopSped", "Enter a number") end
        if Y.LoopConn then Y.LoopConn:Disconnect() end
        Y.LoopVal = n
        Y.LoopConn = Run.Heartbeat:Connect(function()
            local h = LP.Character and LP.Character:FindFirstChildOfClass("Humanoid")
            if h and h.WalkSpeed ~= Y.LoopVal then h.WalkSpeed = Y.LoopVal end
        end)
        notify("LoopSped", "Locked: " .. n)
    end,
    Unfunc = function()
        if Y.LoopConn then
            Y.LoopConn:Disconnect()
            Y.LoopConn = nil
        end
        notify("LoopSped", "Disabled")
    end
})

Y:Register("fly", {
    Func = function(a)
        local sp = tonumber(a[1]) or 50
        local c = LP.Character
        if not c then return end
        local hrp = c:FindFirstChild("HumanoidRootPart")
        if not hrp then return end
        if Y.FlyConn then Y.FlyConn:Disconnect() end
        if Y.FlyParts then
            for _, p in ipairs(Y.FlyParts) do p:Destroy() end
        end

        local bv = Instance.new("BodyVelocity")
        bv.MaxForce = Vector3.new(1e5, 1e5, 1e5)
        bv.Velocity = Vector3.zero
        bv.Parent = hrp

        local bg = Instance.new("BodyGyro")
        bg.MaxTorque = Vector3.new(1e5, 1e5, 1e5)
        bg.P = 1000
        bg.Parent = hrp

        Y.FlyConn = Run.RenderStepped:Connect(function()
            local cam = workspace.CurrentCamera
            local m = Vector3.zero
            if UIS:IsKeyDown(Enum.KeyCode.W) then m = m + cam.CFrame.LookVector end
            if UIS:IsKeyDown(Enum.KeyCode.S) then m = m - cam.CFrame.LookVector end
            if UIS:IsKeyDown(Enum.KeyCode.A) then m = m - cam.CFrame.RightVector end
            if UIS:IsKeyDown(Enum.KeyCode.D) then m = m + cam.CFrame.RightVector end
            if UIS:IsKeyDown(Enum.KeyCode.Space) then m = m + Vector3.new(0, 1, 0) end
            if UIS:IsKeyDown(Enum.KeyCode.LeftControl) then m = m - Vector3.new(0, 1, 0) end
            bv.Velocity = m.Magnitude > 0 and m.Unit * sp or Vector3.zero
            bg.CFrame = cam.CFrame
        end)

        Y.FlyParts = {bv, bg}
        notify("Fly", "Enabled")
    end,
    Unfunc = function()
        if Y.FlyConn then Y.FlyConn:Disconnect() end
        if Y.FlyParts then
            for _, p in ipairs(Y.FlyParts) do p:Destroy() end
        end
        Y.FlyConn = nil
        Y.FlyParts = nil
        notify("Fly", "Disabled")
    end
})

Y:Register("tp", {
    Aliases = {"teleport"},
    Func = function(a)
        local q = table.concat(a, " ")
        if q == "" then return notify("TP", "Enter player name, random, all or me") end
        local low = q:lower()

        if low == "all" then
            local count = 0
            for _, p in ipairs(Players:GetPlayers()) do
                if p ~= LP and p.Character then
                    local hrp = p.Character:FindFirstChild("HumanoidRootPart")
                    local my = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
                    if hrp and my then
                        my.CFrame = hrp.CFrame + Vector3.new(0, 3, 0)
                        count = count + 1
                        task.wait(0.1)
                    end
                end
            end
            notify("TP", "Teleported to " .. count .. " players")
            return
        end

        local t = Y:FindPlayer(q)
        if not t then return notify("TP", "Not found: " .. q) end
        if not t.Character then return notify("TP", t.Name .. " has no character") end
        local hrp = t.Character:FindFirstChild("HumanoidRootPart")
        local my = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
        if hrp and my then
            my.CFrame = hrp.CFrame + Vector3.new(0, 3, 0)
            notify("TP", "To " .. t.Name)
        end
    end
})
